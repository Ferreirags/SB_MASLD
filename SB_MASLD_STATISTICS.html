  select(where(is.numeric), -SEQN, -WTSAFPRP, -SDMVPSU, -SDMVSTRA)

par(mfrow = c(2, 2))

normalidade <- c()
for (a in 1:length(df_aux)) {
  col_data <- df_aux[[a]] # Seleciona a coluna atual
  MEAN <- mean(col_data, na.rm = TRUE)  # Calcula a média da coluna, removendo NAs
  normalidade[[a]] <- nortest::ad.test(col_data)
  col_name <- colnames(df_aux)[a]
  print(paste("coluna: ", a, col_name))
  print(ad.test(col_data))
  hist(col_data, main = col_name, xlab = col_name)
  abline(v = MEAN, lwd = 3, col = "red") # Adiciona a linha média
  mtext(ifelse(normalidade[[a]]$p.value < 0.05, paste("p-value < 0.05 "), paste("p-value > 0.05 ")), side = 3, adj = 1, cex = 0.8)
}

P-valores são altamente dependentes do tamanho amostral: pequenas diferenças podem ser estatisticamente significativas em grandes amostras (sem relevância clínica), enquanto diferenças importantes podem não ser detectadas em amostras pequenas.

Confundimento não deve ser avaliado com base em p-valores, mas sim com base em conhecimento prévio e em análises ajustadas.

Em estudos randomizados, diferenças entre grupos são atribuídas ao acaso, tornando os p-valores ainda mais irrelevantes.

# Função para plotar e salvar os gráficos
plot_and_save <- function(data, y_var, x_vars, output_dir = here("Figures", "Univariate_correlations")) {
  # Criar diretório de saída, se não existir
  if (!dir.exists(output_dir)) {
    dir.create(output_dir)
  }

  for (x_var in x_vars) {
    x <- data[[x_var]]
    y <- data[[y_var]]

    # Nome do arquivo de saída
    file_name <- paste0(output_dir, "/", y_var, "_vs_", x_var, ".png")

    # Ignorar variáveis completamente vazias
    if (all(is.na(x)) | all(is.na(y))) {
      message(paste("Variável ignorada (só contém NAs):", x_var))
      next
    }

    # Salvar o gráfico em um arquivo
    png(file_name, width = 800, height = 600) # Ajuste o tamanho conforme necessário
    if (is.numeric(x)) {
      # Se variável for numérica
      tryCatch({
        plot(
          x, y,
          col = "gray",
          xlab = x_var,
          ylab = y_var,
          las = 1,
          pch = 20,
          font.lab = 2,
          font.axis = 2,
          main = paste(y_var, "vs", x_var)
        )
        abline(lm(y ~ x), lwd = 2, col = "red")
      }, error = function(e) {
        message(paste("Erro ao plotar variável contínua:", x_var, "-", e$message))
      })

    } else {
      # Se variável for categórica
      tryCatch({
        plot(
          as.factor(x), y,
          col = "gray",
          ylab = y_var,
          xlab = x_var,
          las = 1,
          pch = 20,
          font.lab = 2,
          font.axis = 2,
          main = paste(y_var, "vs", x_var)
        )

        MEANS <- tapply(y, x, mean, na.rm = TRUE)
        points(levels(as.factor(x)), MEANS, pch = 20, cex = 3, col = "blue")
      }, error = function(e) {
        message(paste("Erro ao plotar variável categórica:", x_var, "-", e$message))
      })
    }
    dev.off() # Fechar o dispositivo gráfico
  }
}
